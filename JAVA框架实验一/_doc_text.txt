三、知识要点回顾
● MyBatis 是一款优秀的半自动化 ORM 持久层框架，通过 XML 映射文件或注解将 Java 方法与 SQL 语句映射，消除了 JDBC 几乎所有的手动设置参数与结果集封装代码；
● 核心组件：SqlSessionFactory（会话工厂，全局唯一）、SqlSession（会话，非线程安全，用完须关闭）、Mapper 接口（面向接口编程，由框架动态代理生成实现）；
● MyBatis 执行流程：读取配置 → 构建 SqlSessionFactory → 开启 SqlSession → 获取 Mapper 代理 → 执行 SQL → 结果映射 → 返回 Java 对象；
● MyBatis-Plus（MP）在 MyBatis 的基础上只做增强不做改变，内置通用 Mapper（BaseMapper）、条件构造器（Wrapper）与分页插件，实现单表 CRUD 零 SQL 开发；
● #{} 采用 PreparedStatement 预编译占位符（防 SQL 注入，首选）；${} 直接字符串拼接（仅用于动态表名/排序字段等场景，须校验输入）。

步骤 2.3 编写 Mapper 接口与映射文件
package com.lab.mapper;
 
import com.lab.entity.Emp;
import java.util.List;
 
public interface EmpMapper {
    List<Emp> selectAll();
    Emp selectById(Integer empId);
    int insert(Emp emp);
    int update(Emp emp);
    int deleteById(Integer empId);
}
<?xml version="1.0" encoding="UTF-8"?>
<!DOCTYPE mapper PUBLIC
    "-//mybatis.org//DTD Mapper 3.0//EN"
    "http://mybatis.org/dtd/mybatis-3-mapper.dtd">
<mapper namespace="com.lab.mapper.EmpMapper">
 
    <!-- 自定义结果映射：解决字段名与属性名不一致 -->
    <resultMap id="empMap" type="Emp">
        <id property="empId" column="emp_id"/>
        <result property="empName" column="emp_name"/>
        <result property="gender" column="gender"/>
        <result property="dept" column="dept"/>
        <result property="post" column="post"/>
        <result property="salary" column="salary"/>
        <result property="hireDate" column="hire_date"/>
        <result property="status" column="status"/>
    </resultMap>
 
    <sql id="allColumns">
        emp_id, emp_name, gender, dept, post, salary, hire_date, status
    </sql>
 
    <select id="selectAll" resultMap="empMap">
        SELECT <include refid="allColumns"/> FROM emp
    </select>
 
    <select id="selectById" parameterType="int" resultMap="empMap">
        SELECT <include refid="allColumns"/> FROM emp WHERE emp_id = #{empId}
    </select>
 
    <!-- 新增并回填自增主键 -->
    <insert id="insert" parameterType="Emp"
            useGeneratedKeys="true" keyProperty="empId">
        INSERT INTO emp (emp_name, gender, dept, post, salary, hire_date, status)
        VALUES (#{empName}, #{gender}, #{dept}, #{post}, #{salary}, #{hireDate}, #{status})
    </insert>
 
    <update id="update" parameterType="Emp">
        UPDATE emp
        SET emp_name = #{empName}, gender = #{gender}, dept = #{dept},
            post = #{post}, salary = #{salary}, status = #{status}
        WHERE emp_id = #{empId}
    </update>
 
    <delete id="deleteById" parameterType="int">
        DELETE FROM emp WHERE emp_id = #{empId}
    </delete>
</mapper>
步骤 2.4 编写第一个 MyBatis 程序
package com.lab.test;
 
import com.lab.entity.Emp;
import com.lab.mapper.EmpMapper;
import org.apache.ibatis.io.Resources;
import org.apache.ibatis.session.SqlSession;
import org.apache.ibatis.session.SqlSessionFactory;
import org.apache.ibatis.session.SqlSessionFactoryBuilder;
import java.io.InputStream;
import java.util.List;
 
public class MyBatisDemo {
    public static void main(String[] args) throws Exception {
        // 1. 读取配置文件，构建 SqlSessionFactory（全局只需构建一次）
        InputStream in = Resources.getResourceAsStream("mybatis-config.xml");
        SqlSessionFactory factory = new SqlSessionFactoryBuilder().build(in);
        // 2. 开启 SqlSession（用完必须关闭）
        try (SqlSession session = factory.openSession()) {
            // 3. 获取 Mapper 动态代理对象
            EmpMapper mapper = session.getMapper(EmpMapper.class);
            List<Emp> list = mapper.selectAll();
            list.forEach(System.out::println);
        }
    }
}

package com.lab.emp.config;
 
import com.alibaba.druid.pool.DruidDataSource;
import org.mybatis.spring.SqlSessionFactoryBean;
import org.mybatis.spring.mapper.MapperScannerConfigurer;
import org.springframework.context.annotation.*;
import org.springframework.core.io.support.PathMatchingResourcePatternResolver;
import org.springframework.jdbc.datasource.DataSourceTransactionManager;
import org.springframework.transaction.annotation.EnableTransactionManagement;
import javax.sql.DataSource;
 
@Configuration
@ComponentScan(basePackages = "com.lab.emp",
        excludeFilters = @ComponentScan.Filter(type = FilterType.ANNOTATION,
                classes = org.springframework.stereotype.Controller.class))
@PropertySource("classpath:jdbc.properties")
@EnableTransactionManagement
public class SpringConfig {
 
    @Bean
    public DataSource dataSource(
            @Value("${jdbc.driver}") String driver,
            @Value("${jdbc.url}") String url,
            @Value("${jdbc.username}") String user,
            @Value("${jdbc.password}") String pwd) {
        DruidDataSource ds = new DruidDataSource();
        ds.setDriverClassName(driver);
        ds.setUrl(url);
        ds.setUsername(user);
        ds.setPassword(pwd);
        ds.setInitialSize(5);
        ds.setMaxActive(20);
        return ds;
    }
 
    @Bean
    public SqlSessionFactoryBean sqlSessionFactory(DataSource ds) throws Exception {
        SqlSessionFactoryBean fb = new SqlSessionFactoryBean();
        fb.setDataSource(ds);
        // 实体类别名包（MP 实体）
        fb.setTypeAliasesPackage("com.lab.emp.entity");
        // 注册 MyBatis-Plus 分页插件（import com.baomidou.mybatisplus.extension.plugins.
        // MybatisPlusInterceptor 与 ...inner.PaginationInnerInterceptor）
        MybatisPlusInterceptor mp = new MybatisPlusInterceptor();
        mp.addInnerInterceptor(new PaginationInnerInterceptor());
        fb.setPlugins(mp);
        // 开启驼峰映射（使用 MP 提供的 MybatisConfiguration）
        org.apache.ibatis.session.Configuration cfg =
                new com.baomidou.mybatisplus.core.MybatisConfiguration();
        cfg.setMapUnderscoreToCamelCase(true);
        fb.setConfiguration(cfg);
        return fb;
    }
 
    @Bean
    public MapperScannerConfigurer mapperScannerConfigurer() {
        MapperScannerConfigurer cfg = new MapperScannerConfigurer();
        cfg.setBasePackage("com.lab.emp.mapper");
        return cfg;
    }
 
    @Bean
    public DataSourceTransactionManager transactionManager(DataSource ds) {
        return new DataSourceTransactionManager(ds);
    }
}

package com.lab.emp.interceptor;
 
import com.lab.emp.common.Result;
import org.springframework.web.servlet.HandlerInterceptor;
import javax.servlet.http.*;
 
public class LoginInterceptor implements HandlerInterceptor {
    @Override
    public boolean preHandle(HttpServletRequest req, HttpServletResponse resp,
                             Object handler) throws Exception {
        Object user = req.getSession().getAttribute("loginUser");
        if (user == null) {
            resp.setContentType("application/json;charset=UTF-8");
            resp.getWriter().write("{"code":401,"message":"未登录"}");
            return false;
        }
        return true;
    }
}
步骤 6.4 AOP 请求日志切面
package com.lab.emp.aspect;
 
import org.aspectj.lang.ProceedingJoinPoint;
import org.aspectj.lang.annotation.*;
import org.springframework.stereotype.Component;
import java.util.Arrays;
 
@Aspect
@Component
public class WebLogAspect {
 
    @Pointcut("execution(* com.lab.emp.controller..*(..))")
    public void pt() {}
 
    @Around("pt()")
    public Object around(ProceedingJoinPoint pjp) throws Throwable {
        long start = System.currentTimeMillis();
        String method = pjp.getSignature().toShortString();
        System.out.println("[REQ] " + method + " 参数：" + Arrays.toString(pjp.getArgs()));
        try {
            Object result = pjp.proceed();
            System.out.println("[RESP] " + method + " 耗时："
                    + (System.currentTimeMillis() - start) + "ms");
            return result;
        } catch (Throwable t) {
            System.err.println("[ERR] " + method + " 异常：" + t.getMessage());
            throw t;
        }
    }
}
